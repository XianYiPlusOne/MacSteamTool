cmake_minimum_required(VERSION 3.20)
project(FluentManifestPins LANGUAGES C CXX)
if(NOT CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Target macOS architecture")
endif()
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED DOBBY_SOURCE)
  message(FATAL_ERROR "Use script/build_pins.sh to fetch the pinned Dobby source")
endif()
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver ON CACHE BOOL "" FORCE)
add_subdirectory("${DOBBY_SOURCE}" dobby EXCLUDE_FROM_ALL)
include("${CMAKE_CURRENT_SOURCE_DIR}/../Common/dobby_patch.cmake")
add_library(fluent_manifest SHARED runtime.cpp config.cpp)
set_target_properties(fluent_manifest PROPERTIES PREFIX "" INSTALL_NAME_DIR "@loader_path")
target_compile_features(fluent_manifest PRIVATE cxx_std_20)
target_link_libraries(fluent_manifest PRIVATE dobby_static z)
target_link_options(fluent_manifest PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
add_executable(FluentManifestProbe probe.cpp)
target_compile_features(FluentManifestProbe PRIVATE cxx_std_20)
target_link_libraries(FluentManifestProbe PRIVATE fluent_manifest)
